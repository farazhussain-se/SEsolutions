        function applySBBrand() {

            if (sb) {
                document.documentElement.style.setProperty('--primary', sb);
                return;
            }

                if (sb) document.documentElement.style.setProperty('--primary', sb);
            } catch (e) { /* cross-origin, ignore */ }

        function applyTranslations() {

                if (t[key] !== undefined) el.textContent = t[key];

        // --- Render ---
        function renderList() {
            const container = document.getElementById('listContainer');
            container.innerHTML = '';

            const searchTerm = document.getElementById('searchInput').value.toLowerCase();
            const checkedTypes = Array.from(document.querySelectorAll('.checkbox-list input:checked')).map(cb => cb.value);

            let filtered = data.filter(d => d.cat === state.view);

            filtered = filtered.filter(d => {
                const title = (loc(d, 'title') || '').toLowerCase();
                return title.includes(searchTerm) || d.id.toLowerCase().includes(searchTerm);
            });

            if (state.view === 'items') {
                filtered = filtered.filter(d => checkedTypes.some(type => d.type.includes(type)));
            }

            filtered.sort((a, b) => {
                const dateA = new Date(a.updated), dateB = new Date(b.updated);
                const createA = new Date(a.created), createB = new Date(b.created);
                switch (state.sort) {
                    case 'update_new': return dateB - dateA;
                    case 'update_old': return dateA - dateB;
                    case 'create_new': return createB - createA;
                    case 'prio_high':  return a.prio - b.prio;
                    case 'prio_low':   return b.prio - a.prio;
                    default:           return dateB - dateA;
                }
            });

            if (filtered.length === 0) {
                container.innerHTML = `<div class="empty-state">${t.noItems}</div>`;
                return;
            }

            const frag = document.createDocumentFragment();
            filtered.forEach(item => {
                const isHighPrio = item.prio <= 2;
                const prioClass = isHighPrio ? 'priority-high' : 'priority-mod';
                const stateClass = item.state === 'On Hold' ? 'state-hold' : 'state-prog';

                const prioText = t[PRIO_KEY[item.prioLabel]] || item.prioLabel;
                const stateText = t[STATE_KEY[item.state]] || item.state;
                const titleText = loc(item, 'title');

                let footerHtml;
                if (item.by) {
                    footerHtml = tr('openedByTpl', {
                        date: formatDate(item.created),
                        name: `<span class="by-name">${escapeHtml(item.by)}</span>`
                    });
                } else {
                    footerHtml = `${escapeHtml(t.updatedLabel)} ${escapeHtml(formatDate(item.updated))}`;
                }

                const div = document.createElement('div');
                div.className = 'ticket-item';
                div.innerHTML = `
                    <div class="ticket-header">
                        <span class="t-id">${escapeHtml(item.id)}</span>
                        <span class="t-badge">${escapeHtml(item.type)}</span>
                    </div>
                    <div class="t-meta">
                        <span class="${prioClass}">${escapeHtml(prioText)}</span>
                        <span> · </span>
                        <span class="${stateClass}">${escapeHtml(stateText)}</span>
                    </div>
                    <div class="t-title">${escapeHtml(titleText)}</div>
                    <div class="t-footer">${footerHtml}</div>
                `;
                frag.appendChild(div);
            });
            container.appendChild(frag);
        }

        function escapeHtml(s) {
            return String(s ?? '').replace(/[&<>"']/g, c => ({
                '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
            }[c]));
        }

            document.getElementById('viewDropdown').classList.remove('show');
            renderList();
        }

            document.getElementById('contextMenu').classList.toggle('show');
        }

        function toggleNotifications() {
            document.getElementById('notifView').classList.add('active');
        }

        function closeOverlay(id) {
            document.getElementById(id).classList.remove('active');
        }

        function toggleFilterPanel() {
            const panel = document.getElementById('filterPanel');
            const arrow = document.getElementById('filterArrow');
            state.filtersOpen = !state.filtersOpen;
            panel.classList.toggle('open', state.filtersOpen);
            arrow.classList.toggle('up', state.filtersOpen);
        }

        function toggleDropdown(id) {
            document.querySelectorAll('.cw-dropdown').forEach(d => {
                if (d.id !== id) d.classList.remove('show');
            });
            document.getElementById(id).classList.toggle('show');
        }

        function setSort(e, sortType) {
            state.sort = sortType;
            const element = e.currentTarget;
            document.querySelectorAll('.cw-sort-option').forEach(el => el.classList.remove('selected'));
            element.classList.add('selected');
            document.getElementById('sortLabel').textContent = element.textContent.trim();
            document.getElementById('sortDropdown').classList.remove('show');
            renderList();
        }

        function refreshList() {
            const list = document.getElementById('listContainer');
            list.innerHTML = `<div class="loading-row"><span class="spinner"></span>${escapeHtml(t.refreshing)}</div>`;
            setTimeout(renderList, 800);
            document.getElementById('contextMenu').classList.remove('show');
        }

        function openNewTicket() {
            document.getElementById('newTicketView').classList.add('active');
            document.getElementById('contextMenu').classList.remove('show');
        }

        function submitTicket(e) {
            e.preventDefault();
            const typeSel = document.getElementById('ntType');
            const prioSel = document.getElementById('ntPriority');
            const titleInput = document.getElementById('ntTitle');
            const descInput = document.getElementById('ntDescription');

            const type = typeSel.value;
            const prioVal = parseInt(prioSel.value, 10);
            // Map back to canonical English priority label so PRIO_KEY lookup works
            const canonicalPrio = ['', '1 - Critical', '2 - High', '3 - Moderate', '4 - Low'][prioVal] || `${prioVal}`;
            const title = titleInput.value;
            const description = descInput.value;

            const requestedBy = getRequestedByName();

            const newItem = {
                id: (type === 'Incident' ? 'INC' : 'REQ') + String(Math.floor(Math.random() * 1000000)).padStart(7, '0'),
                type: type,
                prio: prioVal,
                prioLabel: canonicalPrio,
                state: 'New',
                title: title,
                description: description,
                updated: new Date().toISOString(),
                created: new Date().toISOString(),
                cat: 'items',
                by: requestedBy
            };

            data.unshift(newItem);
            document.getElementById('ticketForm').reset();
            closeOverlay('newTicketView');

            if (state.view !== 'items') switchMainView('items');
            else renderList();
        }

        // Close menus on outside click
        window.addEventListener('click', function (e) {
            if (!e.target.closest('.view-select')) {
                document.getElementById('viewDropdown').classList.remove('show');
            }
            if (!e.target.closest('.icon-actions') && !e.target.closest('.cw-context-menu')) {
                document.getElementById('contextMenu').classList.remove('show');
            }
            if (!e.target.closest('.dropdown-wrapper')) {
                document.querySelectorAll('.cw-dropdown').forEach(d => d.classList.remove('show'));
            }
        });

        // --- Init ---
        function init() {
            applySBBrand();
            applyTranslations();
            renderList();
        }

        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', init);
        } else {
            init();
        }

        // Expose handlers referenced by inline on* attributes in the mounted markup —
        // this script runs inside `new Function(...)`, so top-level function
        // declarations are local to that call and would otherwise be invisible
        // to onclick="..." lookups, which resolve names against `window`.
        window.toggleViewDropdown = toggleViewDropdown;
        window.switchMainView = switchMainView;
        window.toggleContextMenu = toggleContextMenu;
        window.toggleNotifications = toggleNotifications;
        window.closeOverlay = closeOverlay;
        window.toggleFilterPanel = toggleFilterPanel;
        window.toggleDropdown = toggleDropdown;
        window.setSort = setSort;
        window.refreshList = refreshList;
        window.openNewTicket = openNewTicket;
        window.submitTicket = submitTicket;
        window.renderList = renderList;
